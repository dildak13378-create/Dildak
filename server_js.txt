import "dotenv/config";
import express from "express";
import OpenAI from "openai";

const app = express();
app.use(express.json({ limit: "1mb" }));
app.use(express.static("public"));

const apiKey = process.env.OPENROUTER_API_KEY;
const model = process.env.OPENROUTER_MODEL || "openrouter/free";
const client = apiKey ? new OpenAI({
  apiKey,
  baseURL: "https://openrouter.ai/api/v1",
  defaultHeaders: {
    "HTTP-Referer": "http://localhost:3000",
    "X-OpenRouter-Title": "ROBLOX GOD AI"
  }
}) : null;

const SYSTEM_PROMPT = `Ты ROBLOX GOD AI — помощник-разработчик Roblox Studio.
Отвечай по-русски. Помогай создавать игры в Roblox Studio на Luau.
Когда пользователь просит код:
- указывай, куда именно вставить Script, LocalScript или ModuleScript;
- учитывай серверную и клиентскую безопасность;
- не доверяй клиенту важные игровые данные и награды;
- объясняй, какие объекты нужно создать в Explorer;
- давай полный код и пошаговые инструкции;
- не утверждай, что изменил Studio или создал объекты, если этого реально не сделал.
Пока плагин Roblox Studio не подключён, ты генерируешь код и инструкции, но не управляешь Studio напрямую.`;

app.post("/api/chat", async (req, res) => {
  try {
    if (!client) return res.status(503).json({ error: "На сервере не задан OPENROUTER_API_KEY." });
    const incoming = Array.isArray(req.body.messages) ? req.body.messages : [];
    const messages = [
      { role: "system", content: SYSTEM_PROMPT },
      ...incoming.slice(-16).map(m => ({
        role: m.role === "assistant" ? "assistant" : "user",
        content: String(m.content || "").slice(0, 12000)
      }))
    ];
    if (messages.length === 1) return res.status(400).json({ error: "Напиши сообщение." });

    const completion = await client.chat.completions.create({
      model,
      messages,
      temperature: 0.7
    });
    res.json({ reply: completion.choices?.[0]?.message?.content || "Модель не вернула текстовый ответ." });
  } catch (err) {
    console.error("OpenRouter request failed:", err?.message || err);
    res.status(500).json({ error: "Ошибка OpenRouter. Проверь ключ, модель и лимиты бесплатного тарифа." });
  }
});

const port = Number(process.env.PORT || 3000);
app.listen(port, () => console.log(`ROBLOX GOD AI запущен на порту ${port}`));
